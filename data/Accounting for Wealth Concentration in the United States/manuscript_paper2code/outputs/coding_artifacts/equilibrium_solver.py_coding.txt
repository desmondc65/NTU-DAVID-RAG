## equilibrium_solver.py
import numpy as np
from scipy import optimize
from typing import Dict, Any, List, Tuple
from config import Config
from household_solver import HouseholdProblem
from simulation import Simulator

class Economy:
    """
    Handles aggregate production logic and government budget accounting.
    """

    def __init__(self, config: Config):
        """
        Initializes the Economy helper.

        Args:
            config: Configuration object containing model parameters.
        """
        self.params = config.params
        self.alpha = self.params.get('alpha', 0.33)
        self.psi = self.params.get('psi', 1.55)
        self.delta_annual = self.params.get('delta_annual', 0.045)
        self.period_len = self.params.get('period_length_years', 5)

    def firm_production(self, K: float, N: float) -> Dict[str, float]:
        """
        Calculates prices and output based on a Cobb-Douglas production function.
        Y = Psi * K^alpha * N^(1-alpha)

        Args:
            K: Aggregate capital.
            N: Aggregate labor (efficiency units).

        Returns:
            Dictionary containing 'Y', 'w' (wage), and 'r' (5-year net interest rate).
        """
        # Ensure positive values for stability in root finding
        K = max(K, 1e-6)
        N = max(N, 1e-6)

        Y = self.psi * (K ** self.alpha) * (N ** (1.0 - self.alpha))
        
        # w = MP_L = Psi * (1-alpha) * (K/N)^alpha
        w = self.psi * (1.0 - self.alpha) * ((K / N) ** self.alpha)
        
        # R_gross = MP_K = Psi * alpha * (N/K)^(1-alpha)
        r_gross_annual = self.psi * self.alpha * ((N / K) ** (1.0 - self.alpha))
        
        # Net annual return
        r_net_annual = r_gross_annual - self.delta_annual
        
        # Convert to 5-year return multiplier: (1+r)^5 - 1
        r_5yr = (1.0 + r_net_annual) ** self.period_len - 1.0
        
        return {
            'Y': Y,
            'w': w,
            'r': r_5yr
        }

    def government_budget(self, tax_rev: float, benefits: float, tr: float, Y: float) -> float:
        """
        Calculates the government budget residual.
        Residual = Taxes - (G + Pensions + Transfers)

        Args:
            tax_rev: Total tax revenue (personal, corporate, sales, estate).
            benefits: Total pension outlays.
            tr: Lump-sum transfer guess.
            Y: Aggregate GDP.

        Returns:
            Budget surplus (positive) or deficit (negative).
        """
        # G = 11.6% of GDP as per paper Section IV.E
        G = 0.116 * Y
        
        # Outlays: G + Pensions + Transfers
        total_outlays = G + benefits + tr
        
        return tax_rev - total_outlays


class EquilibriumSolver:
    """
    Orchestrates the general equilibrium solution for the OLG model.
    Finds prices (w, r) and transfers (Tr) that satisfy market clearing and 
    government budget balance.
    """

    def __init__(self, problem: HouseholdProblem, economy: Economy):
        """
        Initializes the EquilibriumSolver.

        Args:
            problem: Instance of HouseholdProblem to solve agent optimization.
            economy: Instance of Economy to handle aggregate accounting.
        """
        self.hp = problem
        self.economy = economy
        self.params = economy.params
        self.config = Config() # For access to calibration targets

    def compute_excess_demand(self, guess: List[float]) -> List[float]:
        """
        Internal function for the root-finder to evaluate market residuals.

        Args:
            guess: [K_guess, N_guess, Tr_guess]

        Returns:
            [Res_K, Res_N, Res_Budget]
        """
        K_guess, N_guess, Tr_guess = guess
        
        # 1. Update prices from guesses
        prices = self.economy.firm_production(K_guess, N_guess)
        w, r, Y = prices['w'], prices['r'], prices['Y']
        
        # 2. Solve Household Lifecycle (Backward Induction)
        # HP policy functions depend on (w, r, Tr)
        policies = self.hp.solve_lifecycle(w, r, Tr_guess)
        
        # 3. Simulate Stationary Distribution
        sim = Simulator(policies, self.params)
        gamma = sim.compute_stationary_distribution()
        
        # 4. Aggregation and Metrics
        metrics = sim.calculate_metrics(gamma, w, r)
        K_sim = metrics['K']
        N_sim = metrics['N']
        
        # 5. Tax Revenue and Pension Calculation
        # Simulator's calculate_metrics provides aggregated tax and benefit info
        # Here we manually compute total tax revenue for clarity in GE logic
        # In a full implementation, Simulator.calculate_metrics() would return 
        # personal_tax_rev, corporate_tax_rev, sales_tax_rev, estate_tax_rev.
        
        # Simplified aggregation of taxes for demonstration (Sum of all taxes)
        # In practice, these components are aggregated inside sim.calculate_metrics()
        # We'll use a placeholder if not explicitly calculated in Sim logic.
        tax_rev_sim = metrics.get('total_tax_rev', 0.22 * Y) # Placeholder based on 22.2% GDP target
        pensions_sim = metrics.get('total_pensions', 0.05 * Y) # Placeholder
        
        # 6. Budget Residual
        budget_res = self.economy.government_budget(tax_rev_sim, pensions_sim, Tr_guess, Y)
        
        # 7. Market Residuals
        res_k = K_guess - K_sim
        res_n = N_guess - N_sim
        
        return [res_k, res_n, budget_res]

    def solve_ge(self) -> Dict[str, Any]:
        """
        Finds the Stationary General Equilibrium.
        Iterates on (K, N, Tr) until the market clearing conditions and 
        government budget are satisfied.

        Returns:
            Dictionary containing equilibrium quantities, prices, and distribution.
        """
        # 1. Initial Guesses based on calibration targets
        # Target K/Y = 3.0, N = 0.35 (hours)
        # Using placeholder values for a starting point
        initial_K = 10.0
        initial_N = 0.35
        initial_Tr = 0.05 # Approx 2.7% of GDP
        
        initial_guess = [initial_K, initial_N, initial_Tr]
        
        print("Starting General Equilibrium search...")
        
        # 2. Solve for Equilibrium using a root-finder
        # Hybrid method (Broyden/Powell) often works well for macro models
        try:
            sol = optimize.root(
                self.compute_excess_demand, 
                initial_guess, 
                method='hybr', 
                tol=1e-4
            )
            
            if not sol.success:
                print("GE Warning: Root finder did not converge. Using last result.")
            
            eq_K, eq_N, eq_Tr = sol.x
            
        except Exception as e:
            print(f"GE Error: {e}")
            # Fallback to damped fixed-point iteration logic if root-finder fails
            eq_K, eq_N, eq_Tr = self._solve_ge_fixed_point(initial_guess)

        # 3. Finalize results
        prices = self.economy.firm_production(eq_K, eq_N)
        policies = self.hp.solve_lifecycle(prices['w'], prices['r'], eq_Tr)
        sim = Simulator(policies, self.params)
        gamma = sim.compute_stationary_distribution()
        metrics = sim.calculate_metrics(gamma, prices['w'], prices['r'])
        
        return {
            'K': eq_K,
            'N': eq_N,
            'Tr': eq_Tr,
            'w': prices['w'],
            'r': prices['r'],
            'Y': prices['Y'],
            'gamma': gamma,
            'metrics': metrics
        }

    def _solve_ge_fixed_point(self, guess: List[float], max_iter: int = 100, damp: float = 0.1) -> Tuple[float, float, float]:
        """
        Fallback fixed-point iteration for GE solving if root-finding is unstable.
        
        Args:
            guess: Initial guess [K, N, Tr].
            max_iter: Maximum iterations.
            damp: Damping factor (weight on new value).
            
        Returns:
            Equilibrium tuple (K, N, Tr).
        """
        K, N, Tr = guess
        
        for i in range(max_iter):
            # Compute excess demands
            residuals = self.compute_excess_demand([K, N, Tr])
            
            # The residuals are (K_guess - K_sim), etc. 
            # So K_sim = K_guess - res
            K_sim = K - residuals[0]
            N_sim = N - residuals[1]
            
            # Budget adjustment for Tr
            # If budget_res > 0 (surplus), increase Tr
            Tr_sim = Tr + 0.5 * residuals[2] 
            
            # Update with damping
            K_new = (1 - damp) * K + damp * K_sim
            N_new = (1 - damp) * N + damp * N_sim
            Tr_new = (1 - damp) * Tr + damp * Tr_sim
            
            dist = np.linalg.norm([K_new - K, N_new - N, Tr_new - Tr])
            K, N, Tr = K_new, N_new, Tr_new
            
            if dist < 1e-5:
                print(f"Fixed-point converged at iteration {i}")
                break
                
        return K, N, Tr
