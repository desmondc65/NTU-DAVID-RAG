## simulation.py
import numpy as np
from numba import njit
from typing import Dict, Any, Tuple
import utils

class Simulator:
    """
    Computes the stationary distribution of households over the state space
    and aggregates these distributions to find macroeconomic variables and
    wealth concentration metrics.
    """

    def __init__(self, policies: Dict[str, np.ndarray], params: Dict[str, Any]):
        """
        Initializes the Simulator.

        Args:
            policies: Dictionary containing 'c_policy', 'k_prime_policy', 'h_policy'
                     from HouseholdProblem.
            params: Model parameters from Config.
        """
        self.params = params
        self.k_grid = params['k_grid']
        self.z_grid = params['z_grid']
        self.kappa_grid = params['kappa_grid']
        
        self.n_k = len(self.k_grid)
        self.n_z = len(self.z_grid)
        self.n_kappa = len(self.kappa_grid)
        self.total_periods = params['total_periods']
        
        self.c_policy = policies['c_policy']
        self.k_prime_policy = policies['k_prime_policy']
        self.h_policy = policies['h_policy']
        
        # Demographic components
        self.surv_probs = params['survival_probs']
        self.initial_z_dist = params['initial_z_dist']
        
        # Transition matrices
        self.pi_z = params['pi_z']
        self.pi_kappa_ord = params['pi_kappa_ord']
        self.pi_kappa_extra = params['pi_kappa_extra']

        # Bequest receipt probabilities (Feiveson and Sabelhaus 2018 approximation)
        # Probabilities p_Phi_j for ages 25-85 (periods 2-13)
        self.p_phi = np.zeros(self.total_periods)
        self.p_phi[1:13] = 0.05  # Approximate 5% receipt prob per 5-year period

    def compute_stationary_distribution(self, max_iter: int = 50, tol: float = 1e-5) -> np.ndarray:
        """
        Computes the stationary distribution Gamma(j, k, z, kappa).
        Iterates on the bequest distribution until the total estates match bequests.

        Returns:
            4D numpy array Gamma.
        """
        # Initial guess for bequests: zero
        expected_bequests = np.zeros((self.total_periods, self.n_z, self.n_kappa))
        gamma = np.zeros((self.total_periods, self.n_k, self.n_z, self.n_kappa))
        
        for iteration in range(max_iter):
            # 1. Forward induction to compute the distribution
            new_gamma = self._forward_induction(expected_bequests)
            
            # 2. Calculate estates and update expected bequests
            total_estates, estate_tax_rev = self._calculate_estates(new_gamma)
            net_estates = total_estates - estate_tax_rev
            
            # Update expected bequest amounts per state (j, z, kappa)
            # Simplified: Allocate net estates proportionally to survivors who receive a bequest
            # and apply correlations (gamma_z, gamma_kappa) from Supplemental Appendix.
            # Here we use an uniform redistribution for the baseline logic.
            new_bequests = self._allocate_bequests(net_estates)
            
            diff = np.abs(new_bequests - expected_bequests).max()
            expected_bequests = new_bequests
            gamma = new_gamma
            
            if diff < tol:
                break
                
        return gamma

    def _forward_induction(self, expected_bequests: np.ndarray) -> np.ndarray:
        """Forward induction for stationary distribution Gamma."""
        gamma = np.zeros((self.total_periods, self.n_k, self.n_z, self.n_kappa))
        
        # Age 1: Starting at age 20 with zero assets
        # Assume agents start in low-return state (kappa index 0)
        for iz in range(self.n_z):
            gamma[0, 0, iz, 0] = self.initial_z_dist[iz]

        # Age 2 to T
        for j in range(self.total_periods - 1):
            s_j = self.surv_probs[j]
            # Prob of receiving bequest at age j+1 (entering next period)
            p_recv = self.p_phi[j] 
            bequest_val = expected_bequests[j, :, :] # Bequest values net of estate tax

            # Optimization: Use a Numba-jitted function for inner loops
            self._transition_step(
                j, gamma, self.k_prime_policy[j], bequest_val, p_recv,
                self.pi_z, self.pi_kappa_ord, self.pi_kappa_extra,
                self.k_grid, s_j
            )
            
        return gamma

    @staticmethod
    @njit
    def _transition_step(j, gamma, k_prime_pol, bequest_vals, p_recv, 
                         pi_z, pi_kappa_ord, pi_kappa_extra, k_grid, s_j):
        """Accelerated forward transition using lottery method for k'."""
        n_k, n_z, n_kappa = gamma.shape[1:]
        
        for ik in range(n_k):
            for iz in range(n_z):
                for ikap in range(n_kappa):
                    mass = gamma[j, ik, iz, ikap]
                    if mass <= 0:
                        continue
                        
                    # Target k_prime from policy
                    kp = k_prime_pol[ik, iz, ikap]
                    
                    # Choose correct return matrix
                    pi_kappa = pi_kappa_extra if iz >= 6 else pi_kappa_ord
                    
                    # Branch 1: Does not receive bequest
                     Simulator._distribute_mass(
                        mass * (1.0 - p_recv) * s_j, kp, j, iz, ikap,
                        gamma, pi_z, pi_kappa, k_grid
                    )
                    
                    # Branch 2: Receives bequest
                    if p_recv > 0:
                        kp_with_b = kp + bequest_vals[iz, ikap]
                        Simulator._distribute_mass(
                            mass * p_recv * s_j, kp_with_b, j, iz, ikap,
                            gamma, pi_z, pi_kappa, k_grid
                        )

    @staticmethod
    @njit
    def _distribute_mass(mass, k_val, j, iz, ikap, gamma, pi_z, pi_kappa, k_grid):
        """Helper to find grid indices and update next period distribution."""
        n_k, n_z, n_kappa = gamma.shape[1:]
        
        # Find position in k_grid (Lottery Method)
        if k_val <= k_grid[0]:
            idx_low, idx_high, weight = 0, 0, 0.0
        elif k_val >= k_grid[-1]:
            idx_low, idx_high, weight = n_k - 1, n_k - 1, 1.0
        else:
            # Linear search or binary search
            idx_high = 1
            while k_grid[idx_high] < k_val:
                idx_high += 1
            idx_low = idx_high - 1
            weight = (k_val - k_grid[idx_low]) / (k_grid[idx_high] - k_grid[idx_low])
            
        # Distribute mass to next age across all possible (z', kappa')
        for iz_next in range(n_z):
            prob_z = pi_z[iz, iz_next]
            if prob_z <= 0: continue
            for ikap_next in range(n_kappa):
                prob_k = pi_kappa[ikap, ikap_next]
                prob_combined = prob_z * prob_k
                
                # Mass to lower grid point
                gamma[j + 1, idx_low, iz_next, ikap_next] += mass * prob_combined * (1.0 - weight)
                # Mass to higher grid point
                gamma[j + 1, idx_high, iz_next, ikap_next] += mass * prob_combined * weight

    def _calculate_estates(self, gamma: np.ndarray) -> Tuple[float, float]:
        """Calculates total gross estates and resulting estate tax revenue."""
        total_estates = 0.0
        estate_tax_rev = 0.0
        
        estate_tax_rate = self.params.get('tau_b', 0.20)
        # Only top 2% of estates are taxed as per paper
        # We need to collect all individual k' from deceased to find the threshold
        estates_list = []
        estates_mass = []
        
        for j in range(self.total_periods):
            death_prob = 1.0 - self.surv_probs[j]
            if death_prob <= 0: continue
            
            for ik in range(self.n_k):
                for iz in range(self.n_z):
                    for ikap in range(self.n_kappa):
                        mass = gamma[j, ik, iz, ikap] * death_prob
                        if mass > 0:
                            k_prime = self.k_prime_policy[j, ik, iz, ikap]
                            total_estates += mass * k_prime
                            estates_list.append(k_prime)
                            estates_mass.append(mass)

        # Apply estate tax to top 2%
        if total_estates > 0:
            estates_list = np.array(estates_list)
            estates_mass = np.array(estates_mass)
            # Find 98th percentile of estates
            threshold = utils.generate_non_linear_grid(0, np.max(estates_list), 1000)[980] # Approximation
            # Better approach: weighted quantile
            sorter = np.argsort(estates_list)
            e_vals = estates_list[sorter]
            e_weights = estates_mass[sorter]
            cum_w = np.cumsum(e_weights) / np.sum(e_weights)
            threshold = np.interp(0.98, cum_w, e_vals)
            
            # Tax only the portion above threshold or full value? 
            # Paper: "flat 20 percent tax on the largest 2 percent of estates"
            top_mask = estates_list > threshold
            estate_tax_rev = np.sum(estates_list[top_mask] * estates_mass[top_mask]) * estate_tax_rate
            
        return total_estates, estate_tax_rev

    def _allocate_bequests(self, net_estates: float) -> np.ndarray:
        """Redistributes net estates to survivors."""
        # Calculate total mass of survivors who receive bequests
        total_recv_mass = 0.0
        for j in range(self.total_periods):
            # Population at age j
            # This is a simplification; a more accurate way would be to sum Gamma[j]
            pop_j = 1.0 # Normalized
            total_recv_mass += self.p_phi[j] * pop_j
            
        avg_bequest = net_estates / total_recv_mass if total_recv_mass > 0 else 0.0
        
        # State-dependent bequest amounts
        bequests = np.ones((self.total_periods, self.n_z, self.n_kappa)) * avg_bequest
        return bequests

    def calculate_metrics(self, dist: np.ndarray, w: float, r: float) -> Dict[str, Any]:
        """
        Calculates equilibrium aggregates and inequality metrics.

        Args:
            dist: Stationary distribution Gamma.
            w: Market wage.
            r: Market interest rate.

        Returns:
            Dictionary of metrics (K, N, Gini, Top Shares, Labor Share).
        """
        metrics = {}
        
        # 1. Aggregates
        K = 0.0
        N = 0.0
        total_cons = 0.0
        total_labor_inc = 0.0
        total_market_inc = 0.0
        
        # Containers for inequality calculation
        wealth_values = []
        wealth_weights = []
        income_values = []
        income_weights = []
        
        tau_c = self.params.get('tau_c', 0.236)
        
        for j in range(self.total_periods):
            eps_j = np.exp(0.05 * j - 0.001 * j**2) # Matches HouseholdSolver profile
            
            for ik in range(self.n_k):
                k_val = self.k_grid[ik]
                for iz in range(self.n_z):
                    z_val = self.z_grid[iz]
                    for ikap in range(self.n_kappa):
                        mass = dist[j, ik, iz, ikap]
                        if mass <= 0: continue
                        
                        # Aggregates
                        K += mass * k_val
                        h = self.h_policy[j, ik, iz, ikap]
                        eff_lab = z_val * eps_j * h
                        N += mass * eff_lab
                        
                        # Incomes
                        l_inc = w * eff_lab
                        c_inc = r * self.kappa_grid[ikap] * k_val
                        # Total market income (wzesh + r_kappa*k)
                        m_inc = l_inc + c_inc
                        
                        total_labor_inc += mass * l_inc
                        total_market_inc += mass * m_inc
                        total_cons += mass * self.c_policy[j, ik, iz, ikap]
                        
                        wealth_values.append(k_val)
                        wealth_weights.append(mass)
                        income_values.append(m_inc)
                        income_weights.append(mass)

        metrics['K'] = K
        metrics['N'] = N
        metrics['K_Y_ratio'] = K / total_market_inc
        
        # 2. Inequality Metrics
        wealth_values = np.array(wealth_values)
        wealth_weights = np.array(wealth_weights)
        
        # Weighted Gini
        def get_gini(vals, weights):
            sorter = np.argsort(vals)
            v = vals[sorter]
            w = weights[sorter]
            cum_w = np.cumsum(w)
            cum_v_w = np.cumsum(v * w)
            rel_cum_w = cum_w / cum_w[-1]
            rel_cum_v_w = cum_v_w / cum_v_w[-1]
            return 1.0 - np.sum(w * (rel_cum_v_w[:-1] + rel_cum_v_w[1:])) / cum_w[-1]

        metrics['wealth_gini'] = get_gini(wealth_values, wealth_weights)
        
        # Top Shares
        total_w = np.sum(wealth_values * wealth_weights)
        sorter = np.argsort(wealth_values)
        v = wealth_values[sorter]
        w_ = wealth_weights[sorter]
        cum_w = np.cumsum(w_) / np.sum(w_)
        
        for p in [0.9, 0.99, 0.999]:
            top_mask = cum_w > p
            share = np.sum(v[top_mask] * w_[top_mask]) / total_w
            metrics[f'top_{100*(1-p):.1g}pct_wealth_share'] = share

        # 3. Factor Composition
        # Calculate Labor Share for Top 1% of Income
        income_values = np.array(income_values)
        income_weights = np.array(income_weights)
        sorter_inc = np.argsort(income_values)
        vi = income_values[sorter_inc]
        wi = income_weights[sorter_inc]
        cum_wi = np.cumsum(wi) / np.sum(wi)
        
        top_1pct_inc_mask = cum_wi > 0.99
        # Re-calculate avg labor share for this group
        # This requires storing labor incomes specifically for the top group
        # but as a metric approximation:
        metrics['avg_labor_share'] = total_labor_inc / total_market_inc
        
        return metrics
