## equilibrium_solver.py
import numpy as np
import pandas as pd
from typing import Dict, Any, Tuple
from config import Config
from params_generator import ParamsGenerator
from household_solver import HouseholdSolver
from simulation_engine import SimulationEngine


class EquilibriumSolver:
    """Orchestrates the general equilibrium (GE) loop for the OLG model.

    This class finds the steady-state equilibrium where factor prices (r, w),
    government policy (pension replacement rate, insurance premium rate),
    and bequests balance market clearing conditions and budget constraints.
    """

    def __init__(self, config: Config):
        """Initializes the equilibrium solver with model configuration.

        Args:
            config: Config object containing model parameters and grid settings.
        """
        self.config = config
        self.params = config.get_all_params()
        self.grid_settings = config.get_grid_settings()
        self.pg = ParamsGenerator(config)

        # Technology constants
        self.alpha = self.params['alpha']
        self.psi_tfp = self.params['psi_tfp']
        self.delta_p = self.params['delta_period']

        # Targets from config
        self.target_pension_gdp = self.params['target_pension_to_gdp']
        self.g_to_gdp = self.params['g_to_gdp']

    def check_market_clearing(self, prices: Dict[str, float], aggregates: Dict[str, float]) -> float:
        """Calculates the residual (excess demand) for the capital market.

        Args:
            prices: Dictionary containing current 'r' and 'w'.
            aggregates: Dictionary containing supply of 'K' and 'N'.

        Returns:
            The absolute difference between capital demand and capital supply.
        """
        r = prices['r']
        k_supply = aggregates['K_supply']
        n_supply = aggregates['N_supply']

        # K_demand / N_demand = ( (r + delta) / (alpha * psi) ) ^ (1 / (alpha - 1))
        kn_ratio_demand = ((r + self.delta_p) / (self.alpha * self.psi_tfp)) ** (1.0 / (self.alpha - 1.0))
        k_demand = kn_ratio_demand * n_supply

        return k_demand - k_supply

    def solve_ge(self, initial_r: float = 0.02, initial_t: float = 0.1, 
                 scenario_params: Dict[str, Any] = None) -> Dict[str, Any]:
        """Iteratively solves for the general equilibrium.

        Args:
            initial_r: Initial guess for the annual interest rate.
            initial_t: Initial guess for the insurance premium rate.
            scenario_params: Optional overrides for counterfactual experiments 
                             (e.g., {'replacement_rate': 0.0} for No Pension).

        Returns:
            A dictionary containing equilibrium prices, aggregates, and government balance.
        """
        # 1. Convert initial annual interest rate guess to period rate
        r_period = (1.0 + initial_r) ** self.params['period_length'] - 1.0
        t_rate = initial_t
        psi_replacement = self.params.get('psi_replacement_guess', 0.4)
        bequest_dist = 0.0

        # Iteration parameters
        tol = self.params['tolerance_ge']
        max_iter = self.params['max_iter_ge']
        damping = 0.3 # Relaxation factor to prevent oscillations

        # Generate exogenous stochastic matrices and deterministic profiles
        z_states, pi_z, _ = self.pg.generate_productivity_matrix("male")
        # Assume female productivity matrix is identical structure for this baseline implementation
        pi_joint = self.pg.get_joint_productivity_transition(pi_z, pi_z)
        h_trans = self.pg.generate_health_transitions()
        omegas = self.pg.calculate_survival_probs()
        eps_profile = self.pg.get_deterministic_wages()

        print(f"Starting GE loop (max {max_iter} iterations)...")

        for iteration in range(1, max_iter + 1):
            # A. Derive Wage from current Interest Rate (Firm FOC)
            # w = (1 - alpha) * psi * ( (r + delta) / (alpha * psi) ) ^ (alpha / (alpha - 1))
            kn_ratio = ((r_period + self.delta_p) / (self.alpha * self.psi_tfp)) ** (1.0 / (self.alpha - 1.0))
            w_rate = (1.0 - self.alpha) * self.psi_tfp * (kn_ratio ** self.alpha)

            # B. Solve Household Decisions
            prices = {'r': r_period, 'w': w_rate}
            
            # Setup current gov policy
            gov_policy = {
                'tau_c': self.params['tau_c'],
                'phi_copay': self.params['phi_copay'],
                'premium_rate': t_rate,
                'replacement_rate': psi_replacement,
                'bequest_dist': bequest_dist,
                'tax_single': self.params['tax_single'],
                'tax_married': self.params['tax_married']
            }
            # Apply counterfactual overrides if provided
            if scenario_params:
                gov_policy.update(scenario_params)

            h_solver = HouseholdSolver(prices, gov_policy, self.params, self.grid_settings)
            
            # backward_induction returns (V_funcs, policy_funcs)
            # For brevity, we assume HS returns a tuple of policy grids required by SimulationEngine
            # The structure used: (ks_m, ks_f, km, ns_m, ns_f, nm, nf, ms_m, ms_f, mm)
            # In a real implementation, HS would solve these and pack them.
            _, _, policies = h_solver.backward_induction(z_states, pi_z, pi_joint, h_trans, omegas, eps_profile)
            
            # C. Simulate Population and Compute Aggregates
            sim_engine = SimulationEngine(self.config, self.pg, policies)
            sim_data = sim_engine.simulate_population(
                N=self.params['num_agents'],
                r=r_period,
                w=w_rate,
                pi_rate=gov_policy['premium_rate'],
                bequest=gov_policy['bequest_dist']
            )
            
            aggregates = sim_engine.compute_aggregates(sim_data)
            
            # D. Check Convergence and Update
            k_supply = aggregates['K_supply']
            n_supply = aggregates['N_supply']
            y_output = aggregates['Output']
            
            # Residual 1: Capital Market
            r_implied = self.alpha * self.psi_tfp * ((k_supply / n_supply) ** (self.alpha - 1.0)) - self.delta_p
            
            # Residual 2: Pension Target (Replacement rate update)
            # Total Pension Exp = psi * Sum(e_tilde)
            # Replacement rate psi must be such that PensionExp / Y = target_pension_gdp
            total_e_tilde_sum = sim_data[sim_data['age_period'] >= self.params['jr']]['earnings'].sum() # Placeholder for benefits base
            if total_e_tilde_sum > 0:
                psi_implied = (self.target_pension_gdp * y_output * (self.params['num_agents'] / 15)) / total_e_tilde_sum
            else:
                psi_implied = psi_replacement

            # Residual 3: Government Budget (Premium rate update)
            # G + SS + MedCoverage = Tax_inc + Tax_cons + PremiumRev
            g_exp = self.g_to_gdp * y_output
            ss_exp = self.target_pension_gdp * y_output
            med_coverage_exp = (1.0 - self.params['phi_copay']) * aggregates['Total_Med_Exp']
            
            tax_inc = (sim_data['earnings'] * 0.1).sum() # Simplified for update logic
            tax_cons = (sim_data['assets'] * 0.05).sum() # Simplified placeholder
            
            needed_from_premiums = (g_exp + ss_exp + med_coverage_exp) - (tax_inc + tax_cons)
            total_labor_income = sim_data[sim_data['age_period'] < self.params['jr']]['earnings'].sum()
            t_implied = needed_from_premiums / total_labor_income if total_labor_income > 0 else t_rate

            # Residual 4: Bequests
            bequest_implied = aggregates['Bequest_Pool'] / (self.params['num_agents'] * (self.params['jr'] - 1) / 15)

            # Check convergence
            diff_r = abs(r_implied - r_period)
            diff_t = abs(t_implied - t_rate)
            diff_psi = abs(psi_implied - psi_replacement)
            
            print(f"Iter {iteration}: r={r_period:.4f}, t={t_rate:.4f}, psi={psi_replacement:.4f} | diff_r={diff_r:.6f}")

            if diff_r < tol and diff_t < tol and diff_psi < tol:
                print(f"GE Converged in {iteration} iterations.")
                return {
                    'r_period': r_period,
                    'w_rate': w_rate,
                    't_rate': t_rate,
                    'psi_replacement': psi_replacement,
                    'bequest_dist': bequest_dist,
                    'aggregates': aggregates,
                    'sim_data': sim_data
                }

            # Update with damping
            r_period = damping * r_implied + (1.0 - damping) * r_period
            t_rate = damping * t_implied + (1.0 - damping) * t_rate
            psi_replacement = damping * psi_implied + (1.0 - damping) * psi_replacement
            bequest_dist = damping * bequest_implied + (1.0 - damping) * bequest_dist

            # Enforce non-negativity
            r_period = max(r_period, 0.001)
            t_rate = max(t_rate, 0.0)
            psi_replacement = max(psi_replacement, 0.0)

        print("Warning: GE solver did not reach full convergence.")
        return {
            'r_period': r_period,
            'w_rate': w_rate,
            't_rate': t_rate,
            'psi_replacement': psi_replacement,
            'bequest_dist': bequest_dist,
            'aggregates': aggregates,
            'sim_data': sim_data
        }


if __name__ == "__main__":
    # Test script for EquilibriumSolver
    cfg = Config("config.yaml")
    solver = EquilibriumSolver(cfg)
    
    # Run a simplified GE solve
    try:
        results = solver.solve_ge(initial_r=0.02, initial_t=0.15)
        print("Equilibrium Factor Prices:")
        print(f"  Interest Rate (period): {results['r_period']:.4f}")
        print(f"  Wage Rate: {results['w_rate']:.4f}")
        print(f"  Premium Rate: {results['t_rate']:.4f}")
    except Exception as e:
        print(f"Solver failed: {e}")
