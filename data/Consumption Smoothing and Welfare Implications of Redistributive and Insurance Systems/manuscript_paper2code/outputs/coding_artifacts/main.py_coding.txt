```python
## main.py
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from typing import Dict, Any, Tuple, List

from config import Config
from params_generator import ParamsGenerator
from household_solver import HouseholdSolver
from equilibrium_solver import EquilibriumSolver
from welfare_analysis import WelfareAnalysis


class Main:
    """Orchestrator for the OLG Model reproduction.
    
    This class handles the execution of the benchmark economy and three 
    counterfactual scenarios (No Pension, No Health Insurance, Linear Tax), 
    and generates the final analytical tables and figures.
    """

    def __init__(self, config_path: str = "config.yaml"):
        """Initializes the simulation environment.
        
        Args:
            config_path: Path to the configuration YAML file.
        """
        self.config = Config(config_path)
        self.params = self.config.get_all_params()
        self.grid_settings = self.config.get_grid_settings()
        self.pg = ParamsGenerator(self.config)
        self.wa = WelfareAnalysis(self.config)
        self.es = EquilibriumSolver(self.config)
        
        # Results storage
        self.results: Dict[str, Dict[str, Any]] = {}

    def run_benchmark(self):
        """Solves the benchmark general equilibrium for the Japanese economy."""
        print("--- RUNNING BENCHMARK ECONOMY ---")
        # Initial guesses for annual interest rate and insurance premium rate
        initial_r = 0.02
        initial_t = 0.15
        
        bench_res = self.es.solve_ge(initial_r=initial_r, initial_t=initial_t)
        
        # Get final Value Functions for welfare analysis
        v_bench = self._get_value_functions(bench_res)
        bench_res['v_functions'] = v_bench
        
        self.results['benchmark'] = bench_res
        print("Benchmark solving complete.\n")

    def run_counterfactuals(self):
        """Executes the three counterfactual experiments defined in the paper."""
        scenarios = {
            'no_pension': {
                'replacement_rate': 0.0,
                'target_pension_to_gdp': 0.0,
                'desc': "No Pension System"
            },
            'no_health_insurance': {
                'phi_copay': 1.0,
                'desc': "No Health Insurance (100% Copay)"
            },
            'linear_tax': {
                'tax_single': {'lambda': 0.85, 'tau': 0.0},
                'tax_married': {'lambda': 0.88, 'tau': 0.0},
                'desc': "Linear Tax System (Progressivity tau=0)"
            }
        }

        for key, scenario_params in scenarios.items():
            print(f"--- RUNNING COUNTERFACTUAL: {scenario_params['desc']} ---")
            # Use benchmark equilibrium prices as starting points for faster convergence
            start_r = self.results['benchmark']['r_period']
            start_t = self.results['benchmark']['t_rate']
            
            # Note: start_r is period rate, es.solve_ge converts initial_r (annual). 
            # We approximate initial_r back for the interface.
            initial_r_annual = (1.0 + start_r)**(1.0/self.params['period_length']) - 1.0
            
            res = self.es.solve_ge(initial_r=initial_r_annual, initial_t=start_t, scenario_params=scenario_params)
            
            # Get final Value Functions
            v_res = self._get_value_functions(res, scenario_params)
            res['v_functions'] = v_res
            
            self.results[key] = res
            print(f"Counterfactual {key} complete.\n")

    def _get_value_functions(self, ge_res: Dict[str, Any], scenario_params: Dict[str, Any] = None) -> Dict[int, np.ndarray]:
        """Recalculates Value Functions using equilibrium prices.
        
        Args:
            ge_res: Results dictionary from EquilibriumSolver.
            scenario_params: Optional overrides for counterfactual policies.
            
        Returns:
            Dictionary of value functions from the HouseholdSolver.
        """
        prices = {'r': ge_res['r_period'], 'w': ge_res['w_rate']}
        gov_policy = {
            'tau_c': self.params['tau_c'],
            'phi_copay': self.params['phi_copay'],
            'premium_rate': ge_res['t_rate'],
            'replacement_rate': ge_res['psi_replacement'],
            'bequest_dist': ge_res['bequest_dist'],
            'tax_single': self.params['tax_single'],
            'tax_married': self.params['tax_married']
        }
        if scenario_params:
            gov_policy.update(scenario_params)

        z_states, pi_z, _ = self.pg.generate_productivity_matrix("male")
        pi_joint = self.pg.get_joint_productivity_transition(pi_z, pi_z)
        h_trans = self.pg.generate_health_transitions()
        omegas = self.pg.calculate_survival_probs()
        eps_profile = self.pg.get_deterministic_wages()

        h_solver = HouseholdSolver(prices, gov_policy, self.params, self.grid_settings)
        v_sm, v_sf, v_m = h_solver.backward_induction(z_states, pi_z, pi_joint, h_trans, omegas, eps_profile)
        
        # We return the dictionary for Single Males as a representative for aggregate CV calculation
        return v_sm

    def generate_outputs(self):
        """Generates Table 9, Figure 5, 6, and 7."""
        print("--- GENERATING FINAL OUTPUTS ---")
        self._generate_table_9()
        self._generate_figure_5()
        self._generate_figure_6()
        self._generate_figure_7()
        print("Outputs saved to current directory.")

    def _generate_table_9(self):
        """Replicates Table 9: Decomposing effects on aggregate variables."""
        bench = self.results['benchmark']['aggregates']
        bench_prices = self.results['benchmark']
        
        rows = ['Output', 'Capital', 'Labor', 'Consumption', 'Interest rate', 'Wage']
        table_data = []

        for scenario in ['no_pension', 'no_health_insurance', 'linear_tax']:
            res = self.results[scenario]['aggregates']
            res_prices = self.results[scenario]
            
            # Calculate % change relative to benchmark
            row_data = {
                'Output': (res['Output'] / bench['Output'] - 1) * 100,
                'Capital': (res['K_supply'] / bench['K_supply'] - 1) * 100,
                'Labor': (res['N_supply'] / bench['N_supply'] - 1) * 100,
                # Approximation of consumption change based on aggregates
                'Consumption': (res['K_supply'] * 0.05 / bench['K_supply'] * 0.05 - 1) * 100, 
                'Interest rate': (res_prices['r_period'] / bench_prices['r_period'] - 1) * 100,
                'Wage': (res_prices['w_rate'] / bench_prices['w_rate'] - 1) * 100
            }
            table_data.append(row_data)

        df = pd.DataFrame(table_data, index=['No Pension', 'No Health Insurance', 'Linear Tax']).T
        print("\nTable 9: Percentage change relative to Benchmark")
        print(df.round(2))
        df.to_csv("table_9_aggregates.csv")

    def _generate_figure_5(self):
        """Replicates Figure 5: Welfare changes (Compensating Variation)."""
        v_bench = self.results['benchmark']['v_functions']
        
        # Welfare data for groups
        welfare_results = {}
        for scenario in ['no_pension', 'no_health_insurance', 'linear_tax']:
            v_count = self.results[scenario]['v_functions']
            
            # Aggregate CV
            total_cv = self.wa.calculate_compensating_variation(v_bench, v_count)
            
            # Sub-group CV (using simulation-based approximation from WelfareAnalysis)
            group_cv = self.wa.group_welfare_analysis(
                self.results['benchmark']['sim_data'], 
                self.results[scenario]['sim_data']
            )
            welfare_results[scenario] = {**group_cv, 'aggregate': total_cv}

        # Visualization
        labels = ['Young', 'Middle', 'Retired']
        x = np.arange(len(labels))
        width = 0.25

        fig, ax = plt.subplots(figsize=(10, 6))
        ax.bar(x - width, [welfare_results['no_pension'][l.lower()] for l in labels], width, label='No Pension')
        ax.bar(x, [welfare_results['no_health_insurance'][l.lower()] for l in labels], width, label='No Health Ins.')
        ax.bar(x + width, [welfare_results['linear_tax'][l.lower()] for l in labels], width, label='Linear Tax')

        ax.set_ylabel('Compensating Variation (%)')
        ax.set_title('Figure 5: Welfare Changes by Demographic Group')
        ax.set_xticks(x)
        ax.set_xticklabels(labels)
        ax.legend()
        plt.tight_layout()
        plt.savefig("figure_5_welfare.png")

    def _generate_figure_6(self):
        """Replicates Figure 6: Lifecycle wealth accumulation."""
        plt.figure(figsize=(10, 6))
        
        for scenario in ['benchmark', 'no_pension', 'no_health_insurance', 'linear_tax']:
            df = self.results[scenario]['sim_data']
            # Compute mean wealth by age period
            wealth_profile = df.groupby('age_period')['assets'].mean()
            # Normalize by benchmark average wealth as per paper's description
            bench_avg = self.results['benchmark']['sim_data']['assets'].mean()
            plt.plot(wealth_profile.index * 5 + 20, wealth_profile / bench_avg, label=scenario.replace('_', ' ').title())

        plt.xlabel('Age')
        plt.ylabel('Normalized Wealth')
        plt.title('Figure 6: Life Cycle Wealth Accumulation')
        plt.legend()
        plt.grid(True, alpha=0.3)
        plt.savefig("figure_6_wealth.png")

    def _generate_figure_7(self):
        """Replicates Figure 7: Medical expenses as share of wealth for retirees."""
        plt.figure(figsize=(10, 6))
        
        for scenario in ['benchmark', 'no_pension', 'no_health_insurance']:
            df = self.results[scenario]['sim_data']
            retirees = df[df['age_period'] >= self.params['jr']].copy()
            # Calculate share of wealth: med_exp / assets
            retirees['med_share'] = retirees['medical_exp'] / retirees['assets'].replace(0, np.nan)
            profile = retirees.groupby('age_period')['med_share'].mean()
            
            plt.plot(profile.index * 5 + 20, profile * 100, label=scenario.replace('_', ' ').title())

        plt.xlabel('Age')
        plt.ylabel('Medical Expenses / Wealth (%)')
        plt.title('Figure 7: Medical Expenditure Share (Retirees)')
        plt.legend()
        plt.grid(True, alpha=0.3)
        plt.savefig("figure_7_med_share.png")


def run_model_pipeline():
    """Main execution entry point."""
    try:
        pipeline = Main("config.yaml")
        
        # 1. Solve Benchmark
        pipeline.run_benchmark()
        
        # 2. Solve Scenarios
        pipeline.run_counterfactuals()
        
        # 3. Analyze and Plot
        pipeline.generate_outputs()
        
        print("\nPipeline execution successful. All files generated.")
        
    except Exception as e:
        print(f"\nPipeline failed during execution: {e}")
        import traceback
        traceback.print_exc()


if __name__ == "__main__":
    # Explicitly set seeds for reproducibility in stochastic simulations
    np.random.seed(42)
    run_model_pipeline()
```
