import numpy as np
import pandas as pd
from numba import njit, prange
from typing import Dict, Any, Tuple
from config import Config
from params_generator import ParamsGenerator

@njit
def interp1d_numba(x_val: float, x_grid: np.ndarray, y_vals: np.ndarray) -> float:
    """Linear interpolation optimized for Numba."""
    if x_val <= x_grid[0]:
        return y_vals[0]
    if x_val >= x_grid[-1]:
        return y_vals[-1]
    
    # Binary search
    low = 0
    high = len(x_grid) - 2
    idx = 0
    while low <= high:
        mid = (low + high) // 2
        if x_grid[mid] <= x_val:
            idx = mid
            low = mid + 1
        else:
            high = mid - 1
            
    weight = (x_val - x_grid[idx]) / (x_grid[idx + 1] - x_grid[idx])
    return y_vals[idx] * (1.0 - weight) + y_vals[idx + 1] * weight

@njit
def get_survival_prob(age_idx: int, gender: int, h: int, omegas: np.ndarray) -> float:
    """Logistic survival probability from paper section 4.2."""
    power = omegas[0] + omegas[1] * age_idx + omegas[2] * (age_idx**2) + omegas[3] * h
    return 1.0 / (1.0 + np.exp(power))

@njit
def simulate_lifecycle_kernel(
    num_agents: int, num_periods: int, jr: int,
    k_grid: np.ndarray, e_grid: np.ndarray, z_grid: np.ndarray, h_grid: np.ndarray,
    policy_ks_m: np.ndarray, policy_ks_f: np.ndarray, policy_km: np.ndarray,
    policy_ns_m: np.ndarray, policy_ns_f: np.ndarray, policy_nm: np.ndarray, policy_nf: np.ndarray,
    policy_ms_m: np.ndarray, policy_ms_f: np.ndarray, policy_mm: np.ndarray,
    pi_z: np.ndarray, pi_joint: np.ndarray, h_trans: np.ndarray,
    omegas: np.ndarray, eps_profile: np.ndarray,
    phi_married_init: float, omega_corr: float,
    r: float, w: float, phi_copay: float, pi_rate: float,
    eta: float, kappa_m: float, kappa_f: float,
    bequest_per_worker: float
) -> Tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
    """
    Core Numba-accelerated simulation loop.
    
    Returns arrays for: assets, labor, medical_exp, earnings, health, marital_status.
    """
    # Outputs: (N_agents, N_periods)
    out_k = np.zeros((num_agents, num_periods))
    out_n = np.zeros((num_agents, num_periods))
    out_m = np.zeros((num_agents, num_periods))
    out_e = np.zeros((num_agents, num_periods)) # Current earnings
    out_h = np.zeros((num_agents, num_periods), dtype=np.int32)
    out_marital = np.zeros((num_agents, num_periods), dtype=np.int32) # 0: S_M, 1: S_F, 2: M
    out_alive = np.ones((num_agents, num_periods), dtype=np.int32)
    
    # Internal states
    curr_k = np.zeros(num_agents)
    curr_e_tilde = np.zeros(num_agents)
    curr_z_idx = np.zeros(num_agents, dtype=np.int32)
    curr_h_idx = np.full(num_agents, 4, dtype=np.int32) # Start with best health
    curr_marital = np.zeros(num_agents, dtype=np.int32)
    is_alive = np.ones(num_agents, dtype=np.int32)

    # Initial assignment (j=1)
    for i in range(num_agents):
        # Marital Status
        if np.random.rand() < phi_married_init:
            curr_marital[i] = 2
        else:
            curr_marital[i] = 0 if np.random.rand() < 0.5 else 1
        
        # Initial Productivity (Simplified draw based on college kappa)
        if curr_marital[i] == 2:
            # Married: draw joint productivity using kappa (Table 3)
            # Simplifying: Draw education then transitory
            edu_draw = np.random.rand()
            if edu_draw < 0.424: # High-High
                curr_z_idx[i] = 5 * 6 + 5 # Simplified joint index
            else:
                curr_z_idx[i] = 1 * 6 + 1
        else:
            # Single: draw 0-5
            curr_z_idx[i] = np.random.randint(0, 6)
            
    # Lifecycle loop
    for j in range(1, num_periods + 1):
        for i in range(num_agents):
            if is_alive[i] == 0:
                continue
            
            age_idx = j
            m_stat = curr_marital[i]
            k_val = curr_k[i]
            et_val = curr_e_tilde[i]
            
            # 1. Decisions based on current state
            if j < jr: # Working age
                if m_stat == 0: # Single Male
                    z_idx = curr_z_idx[i]
                    n = interp1d_numba(et_val, e_grid, policy_ns_m[j-1, :, z_idx, :].flatten()) # Simplified lookup
                    kp = interp1d_numba(et_val, e_grid, policy_ks_m[j-1, :, z_idx, :].flatten())
                    earning = w * np.exp(z_grid[z_idx]) * eps_profile[j-1, 0] * n
                    med = 0.0
                elif m_stat == 1: # Single Female
                    z_idx = curr_z_idx[i]
                    n = interp1d_numba(et_val, e_grid, policy_ns_f[j-1, :, z_idx, :].flatten())
                    kp = interp1d_numba(et_val, e_grid, policy_ks_f[j-1, :, z_idx, :].flatten())
                    earning = w * np.exp(z_grid[z_idx]) * eps_profile[j-1, 1] * n
                    med = 0.0
                else: # Married
                    zj_idx = curr_z_idx[i]
                    zm_idx = zj_idx // 6
                    zf_idx = zj_idx % 6
                    nm = interp1d_numba(et_val, e_grid, policy_nm[j-1, :, zj_idx, :].flatten())
                    nf = interp1d_numba(et_val, e_grid, policy_nf[j-1, :, zj_idx, :].flatten())
                    kp = interp1d_numba(et_val, e_grid, policy_km[j-1, :, zj_idx, :].flatten())
                    earning = (w * np.exp(z_grid[zm_idx]) * eps_profile[j-1, 0] * nm + 
                               w * np.exp(z_grid[zf_idx]) * eps_profile[j-1, 1] * nf) / 2.0
                    n = (nm + nf) / 2.0
                    med = 0.0
                
                # Update Productivity for next period
                if j < jr - 1:
                    if m_stat < 2:
                        # Draw from pi_z[curr_z_idx, :]
                        r_z = np.random.rand()
                        cum_p = 0.0
                        for zp in range(6):
                            cum_p += pi_z[curr_z_idx[i], zp]
                            if r_z < cum_p:
                                curr_z_idx[i] = zp
                                break
                    else:
                        # Married joint draw
                        r_zj = np.random.rand()
                        cum_pj = 0.0
                        for zjp in range(36):
                            cum_pj += pi_joint[curr_z_idx[i], zjp]
                            if r_zj < cum_pj:
                                curr_z_idx[i] = zjp
                                break
            
            else: # Retirement age
                n = 0.0
                earning = 0.0
                h_idx = curr_h_idx[i]
                if m_stat == 0: # Single Male
                    kp = interp1d_numba(et_val, e_grid, policy_ks_m[j-1, :, h_idx, :].flatten())
                    med = interp1d_numba(et_val, e_grid, policy_ms_m[j-1, :, h_idx, :].flatten())
                    s_prob = get_survival_prob(age_idx, 0, h_idx, omegas)
                elif m_stat == 1: # Single Female
                    kp = interp1d_numba(et_val, e_grid, policy_ks_f[j-1, :, h_idx, :].flatten())
                    med = interp1d_numba(et_val, e_grid, policy_ms_f[j-1, :, h_idx, :].flatten())
                    s_prob = get_survival_prob(age_idx, 1, h_idx, omegas)
                else: # Married
                    kp = interp1d_numba(et_val, e_grid, policy_km[j-1, :, h_idx, :].flatten())
                    med = interp1d_numba(et_val, e_grid, policy_mm[j-1, :, h_idx, :].flatten())
                    sm = get_survival_prob(age_idx, 0, h_idx, omegas)
                    sf = get_survival_prob(age_idx, 1, h_idx, omegas)
                    s_prob = sm * sf # Joint survival for next period married state
                    
                    # Check for spousal loss
                    r_die_m = np.random.rand()
                    r_die_f = np.random.rand()
                    if r_die_m > sm and r_die_f <= sf:
                        curr_marital[i] = 1 # Husband died, wife single
                    elif r_die_m <= sm and r_die_f > sf:
                        curr_marital[i] = 0 # Wife died, husband single
                    elif r_die_m > sm and r_die_f > sf:
                        is_alive[i] = 0 # Both died
                
                # Check individual survival
                if m_stat < 2:
                    if np.random.rand() > s_prob:
                        is_alive[i] = 0
                
                # Update Health for next period
                if is_alive[i] == 1 and j < num_periods:
                    cohort = 0 if j < 11 else 1
                    r_h = np.random.rand()
                    cum_h = 0.0
                    for hp in range(5):
                        cum_h += h_trans[cohort, curr_h_idx[i], hp]
                        if r_h < cum_h:
                            curr_h_idx[i] = hp
                            break

            # Store results
            out_k[i, j-1] = k_val
            out_n[i, j-1] = n
            out_m[i, j-1] = med
            out_e[i, j-1] = earning
            out_h[i, j-1] = curr_h_idx[i]
            out_marital[i, j-1] = m_stat
            out_alive[i, j-1] = is_alive[i]
            
            # Step forward
            curr_k[i] = kp
            curr_e_tilde[i] = ((age_idx - 1.0) * et_val + earning) / age_idx
            
    return out_k, out_n, out_m, out_e, out_h, out_marital

class SimulationEngine:
    """Simulates the lifecycle paths of heterogeneous households to compute aggregates."""

    def __init__(self, config: Config, params_gen: ParamsGenerator, policy_functions: Tuple):
        """Initializes the simulation engine with model parameters and policy results.
        
        Args:
            config: Model configuration object.
            params_gen: Parameter generator for transition matrices and survival.
            policy_functions: Tuple containing policy grids for k_prime, n, m.
        """
        self.config = config
        self.params = config.get_all_params()
        self.grid_settings = config.get_grid_settings()
        self.pg = params_gen
        
        # Policy functions: (Type: Dict or Tuple of ndarrays)
        # Expected structure: (ks_m, ks_f, km, ns_m, ns_f, nm, nf, ms_m, ms_f, mm)
        self.policy_ks_m, self.policy_ks_f, self.policy_km, \
        self.policy_ns_m, self.policy_ns_f, self.policy_nm, self.policy_nf, \
        self.policy_ms_m, self.policy_ms_f, self.policy_mm = policy_functions

    def simulate_population(self, N: int = 50000, r: float = 0.02, w: float = 1.0, 
                            pi_rate: float = 0.1, bequest: float = 0.0) -> pd.DataFrame:
        """Runs the simulation for N agents over their lifecycle.
        
        Args:
            N: Number of agents.
            r: Interest rate.
            w: Wage rate.
            pi_rate: Pension insurance premium rate.
            bequest: Total bequest redistributed per worker.

        Returns:
            Pandas DataFrame containing the panel data of the simulated population.
        """
        # Prepare grids and matrices
        z_states, pi_z, _ = self.pg.generate_productivity_matrix("male")
        pi_z_f, _, _ = self.pg.generate_productivity_matrix("female") # Assume same matrix structure
        pi_joint = self.pg.get_joint_productivity_transition(pi_z, pi_z)
        h_trans = self.pg.generate_health_transitions()
        omegas = self.pg.calculate_survival_probs()
        eps_profile = self.pg.get_deterministic_wages()
        
        k_grid = np.linspace(self.grid_settings['k_min'], self.grid_settings['k_max'], self.grid_settings['n_k'])
        e_grid = np.linspace(self.grid_settings['e_min'], self.grid_settings['e_max'], self.grid_settings['n_e'])
        h_grid = np.array(self.grid_settings['h_states'], dtype=np.float64)

        # Run kernel
        k, n, m, e, h, marital = simulate_lifecycle_kernel(
            N, self.params['num_periods'], self.params['jr'],
            k_grid, e_grid, z_states, h_grid,
            self.policy_ks_m, self.policy_ks_f, self.policy_km,
            self.policy_ns_m, self.policy_ns_f, self.policy_nm, self.policy_nf,
            self.policy_ms_m, self.policy_ms_f, self.policy_mm,
            pi_z, pi_joint, h_trans, omegas, eps_profile,
            0.6, self.params['intra_family_corr'], # Assume 60% married initially
            r, w, self.params['phi_copay'], pi_rate,
            self.params['eta'], self.params['kappa_male'], self.params['kappa_female'],
            bequest
        )

        # Reshape to long format for DataFrame
        agent_ids = np.repeat(np.arange(N), self.params['num_periods'])
        ages = np.tile(np.arange(self.params['num_periods']) + 1, N)
        
        data = {
            'agent_id': agent_ids,
            'age_period': ages,
            'assets': k.flatten(),
            'labor_supply': n.flatten(),
            'medical_exp': m.flatten(),
            'earnings': e.flatten(),
            'health_idx': h.flatten(),
            'marital_status': marital.flatten()
        }
        
        df = pd.DataFrame(data)
        # Filter out observations where agents were dead if necessary, 
        # but usually we keep them and handle zeros in aggregates.
        return df

    def compute_aggregates(self, sim_data: pd.DataFrame) -> Dict[str, float]:
        """Calculates macroeconomic variables and government budget items.
        
        Args:
            sim_data: Output from simulate_population.

        Returns:
            Dictionary with aggregate K, N, Y, and government budget components.
        """
        # Constants
        jr = self.params['jr']
        alpha = self.params['alpha']
        psi_tfp = self.params['psi_tfp']
        tau_c = self.params['tau_c']
        phi_copay = self.params['phi_copay']
        
        # Aggregate Capital (Supply)
        K = sim_data['assets'].mean()
        
        # Aggregate Labor (Supply)
        # Weight by productivity is implicit in simulation kernel's earnings logic
        # For pure effective labor units:
        N = sim_data[sim_data['age_period'] < jr]['labor_supply'].mean()
        
        # Output (Firms)
        Y = psi_tfp * (K**alpha) * (N**(1-alpha))
        
        # Government items
        pension_exp = sim_data[sim_data['age_period'] >= jr]['earnings'].sum() # This is a placeholder for SS benefits
        # In actual GE, SS benefits depend on historical average earnings e_tilde
        # Here we approximate for the interface
        
        med_exp_public = (1.0 - phi_copay) * sim_data['medical_exp'].sum()
        
        aggregates = {
            'K_supply': K,
            'N_supply': N,
            'Output': Y,
            'Total_Med_Exp': sim_data['medical_exp'].mean(),
            'Med_Exp_Ratio_GDP': sim_data['medical_exp'].sum() / (Y * len(sim_data) / 15),
            'Bequest_Pool': sim_data[sim_data['assets'] > 0]['assets'].mean() * 0.05 # Simplified death-leakage
        }
        
        return aggregates
