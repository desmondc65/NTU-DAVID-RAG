## evaluation.py
import numpy as np
import matplotlib.pyplot as plt
from typing import Dict, Any, List, Optional
from config import Config

class Evaluation:
    """
    Evaluates the welfare and macroeconomic impacts of policy reforms in the OLG model.

    This class provides tools to calculate Consumption Equivalence Variation (CEV) 
    for different age cohorts and shock levels, and to generate visualizations 
    matching the figures presented in the research paper (Figures 6 and 7).
    """

    def __init__(self, config: Config):
        """
        Initializes the Evaluation module with model parameters.

        Args:
            config (Config): The model configuration object containing preference 
                             and production parameters.
        """
        self.config = config
        self.alpha: float = config.alpha
        self.gamma: float = config.gamma
        self.theta: float = config.theta
        self.mu: float = config.mu
        self.nu: float = config.nu

    def calculate_cev(self, v_base: np.ndarray, v_new: np.ndarray) -> float:
        """
        Calculates the Consumption Equivalence Variation (CEV) as a percentage change.

        According to the paper (Section 4), the uniform percentage change in the 
        baseline consumption path that makes a household indifferent to the 
        policy change is given by:
        lambda = [ (V_new / V_base) ^ (1 / (alpha * (1 - gamma))) ] - 1.

        Args:
            v_base (np.ndarray): Lifetime utility (value function) in the baseline economy.
                                 Can be a scalar or an array of values.
            v_new (np.ndarray): Lifetime utility (value function) after the policy reform.

        Returns:
            float: The percentage consumption equivalent variation (CEV).
        """
        # Ensure values are non-zero to avoid division errors
        # Note: Utilities are negative because gamma > 1
        exponent: float = 1.0 / (self.alpha * (1.0 - self.gamma))
        
        # Calculate CEV (%)
        # We take the mean if arrays are provided to return a single aggregate metric
        v_base_avg = np.mean(v_base)
        v_new_avg = np.mean(v_new)
        
        cev: float = ( (v_new_avg / v_base_avg) ** exponent - 1.0 ) * 100.0
        return cev

    def calculate_cev_grid(self, v_base: np.ndarray, v_new: np.ndarray) -> np.ndarray:
        """
        Calculates CEV across a grid of states (e.g., for Table 5).
        
        This is used to analyze the $5 \times 5$ wage shock grid welfare effects.
        """
        exponent: float = 1.0 / (self.alpha * (1.0 - self.gamma))
        cev_grid: np.ndarray = ( (v_new / v_base) ** exponent - 1.0 ) * 100.0
        return cev_grid

    def plot_macro_results(self, results: Dict[str, Any]) -> None:
        """
        Plots the transition effects of removing spousal and survivors benefits.

        Generates a figure with multiple subplots matching Figure 7 of the paper:
        - Total Output (GDP)
        - Capital Stock (National Wealth)
        - Total Labor Supply (Efficiency Units)
        - Women's Total Work Hours
        - Women's Labor Participation Rate
        - OASI Benefit Adjustment Factor (psi_t)
        - Average Welfare Change by Cohort

        Args:
            results (Dict[str, Any]): Dictionary containing time-paths of 
                                      macroeconomic variables.
        """
        t_path = np.arange(1, self.config.transition_years + 1)
        
        # Extract data paths
        gdp_path = results.get('gdp', np.zeros(self.config.transition_years))
        k_path = results.get('k', np.zeros(self.config.transition_years))
        l_path = results.get('l', np.zeros(self.config.transition_years))
        hours_f_path = results.get('hours_f', np.zeros(self.config.transition_years))
        lfp_f_path = results.get('lfp_f', np.zeros(self.config.transition_years))
        psi_path = results.get('psi', np.ones(self.config.transition_years))
        welfare_path = results.get('welfare_cohorts', np.zeros(self.config.transition_years))

        # Percentage changes from the first year (baseline approximation)
        def to_pct(arr):
            return (arr / arr[0] - 1.0) * 100.0 if arr[0] != 0 else arr

        fig, axes = plt.subplots(3, 2, figsize=(12, 15))
        fig.suptitle('Transition Effects of Removing Spousal and Survivors Benefits', fontsize=16)

        # Plot 1: GDP
        axes[0, 0].plot(t_path, to_pct(gdp_path), 'r--', label='Reform')
        axes[0, 0].set_title('Gross Domestic Product')
        axes[0, 0].set_ylabel('% change from baseline')
        axes[0, 0].grid(True)

        # Plot 2: Capital Stock
        axes[0, 1].plot(t_path, to_pct(k_path), 'r--', label='Reform')
        axes[0, 1].set_title('Capital Stock (National Wealth)')
        axes[0, 1].set_ylabel('% change from baseline')
        axes[0, 1].grid(True)

        # Plot 3: Efficiency Labor
        axes[1, 0].plot(t_path, to_pct(l_path), 'r--', label='Reform')
        axes[1, 0].set_title('Total Labor Supply (Efficiency Units)')
        axes[1, 0].set_ylabel('% change from baseline')
        axes[1, 0].grid(True)

        # Plot 4: Women's Total Hours
        axes[1, 1].plot(t_path, to_pct(hours_f_path), 'r--', label='Reform')
        axes[1, 1].set_title("Women's Total Work Hours")
        axes[1, 1].set_ylabel('% change from baseline')
        axes[1, 1].grid(True)

        # Plot 5: Women's LFP
        axes[2, 0].plot(t_path, to_pct(lfp_f_path), 'r--', label='Reform')
        axes[2, 0].set_title("Women's Labor Participation Rate")
        axes[2, 0].set_ylabel('% change from baseline')
        axes[2, 0].grid(True)

        # Plot 6: OASI Benefit Adjustment (psi_t)
        axes[2, 1].plot(t_path, (psi_path - 1.0) * 100.0, 'r--', label='Reform')
        axes[2, 1].set_title('OASI Benefit Adjustment Factor ($\psi_t$)')
        axes[2, 1].set_ylabel('% change from baseline')
        axes[2, 1].grid(True)

        plt.tight_layout(rect=[0, 0.03, 1, 0.95])
        plt.show()

    def generate_life_cycle_plots(self, 
                                  baseline_means: Dict[str, np.ndarray], 
                                  reform_means: Dict[str, np.ndarray]) -> None:
        """
        Generates life-cycle profile changes matching Figure 6 of the paper.

        Args:
            baseline_means (Dict): Averages by age in the initial steady state.
            reform_means (Dict): Averages by age in the final steady state.
        """
        ages = np.arange(self.config.age_start, self.config.age_end + 1)
        
        metrics = ['consumption', 'wealth', 'hours_m', 'hours_f', 'lfp_f', 'oasi_benefits']
        titles = ['Private Consumption', 'Private Wealth', "Men's Work Hours", 
                  "Women's Work Hours", "Women's Labor Participation", 'OASI Benefits']

        fig, axes = plt.subplots(3, 2, figsize=(12, 12))
        axes = axes.flatten()

        for i, metric in enumerate(metrics):
            b_val = baseline_means.get(metric, np.zeros_like(ages))
            r_val = reform_means.get(metric, np.zeros_like(ages))
            
            # Change calculation
            pct_change = (r_val / b_val - 1.0) * 100.0
            # Handle possible zeros/NaNs in benefits or wealth
            pct_change = np.nan_to_num(pct_change, nan=0.0, posinf=0.0, neginf=0.0)

            axes[i].plot(ages, pct_change, 'b-', linewidth=2)
            axes[i].set_title(titles[i])
            axes[i].set_xlabel('Age')
            axes[i].set_ylabel('% change from baseline')
            axes[i].axhline(0, color='black', linestyle='--', linewidth=0.5)
            axes[i].grid(alpha=0.3)

        plt.tight_layout()
        plt.show()

    def report_table_results(self, baseline_ss: Dict[str, Any], reform_ss: Dict[str, Any]) -> None:
        """
        Prints a summary table of long-run effects matching Table 8 in the paper.
        """
        print("\n" + "="*50)
        print("LONG-RUN EFFECTS OF REMOVING SPOUSAL & SURVIVORS BENEFITS")
        print("="*50)
        print(f"{'Variable':<35} | {'% Change':>10}")
        print("-" * 50)

        # Macro variables
        k_b, l_b = baseline_ss['metrics'].get('k', 1.0), baseline_ss['metrics'].get('l', 1.0)
        k_r, l_r = reform_ss['metrics'].get('k', 1.0), reform_ss['metrics'].get('l', 1.0)
        
        y_b = self.config.A_target * (k_b**self.config.theta) * (l_b**(1.0 - self.config.theta))
        y_r = self.config.A_target * (k_r**self.config.theta) * (l_r**(1.0 - self.config.theta))

        print(f"{'Capital Stock (National Wealth)':<35} | {(k_r/k_b-1)*100:>10.2f}%")
        print(f"{'Labor Supply (Efficiency Units)':<35} | {(l_r/l_b-1)*100:>10.2f}%")
        print(f"{'Gross Domestic Product (GDP)':<35} | {(y_r/y_b-1)*100:>10.2f}%")
        
        # OASI Adjustment
        psi_r = reform_ss['policy_params'].get('psi_t', 1.0)
        print(f"{'OASI Benefit Adjustment (psi)':<35} | {(psi_r-1)*100:>10.2f}%")
        
        # Welfare
        v_b = baseline_ss['value_functions'][0] # Age 21
        v_r = reform_ss['value_functions'][0] # Age 21
        cev = self.calculate_cev(v_b, v_r)
        print(f"{'Welfare of age 21 households (CEV)':<35} | {cev:>10.2f}%")
        print("="*50 + "\n")

    def display_welfare_grid(self, cev_matrix: np.ndarray) -> None:
        """
        Prints the welfare grid for age 21 married couples matching Table 5.
        Rows: Husband's wage levels (e1_1 to e1_5).
        Cols: Wife's wage levels (e2_1 to e2_5).
        """
        print("Welfare changes (CEV %) for age 21 married households by wage level:")
        print("Husband Wage (Row) / Wife Wage (Col)")
        header = "      " + "".join([f"e2_{j+1:<6}" for j in range(5)])
        print(header)
        print("-" * len(header))
        for i in range(5):
            row_str = f"e1_{i+1:<2} |"
            for j in range(5):
                row_str += f"{cev_matrix[i, j]:>7.2f}"
            print(row_str)
        print("-" * len(header) + "\n")
