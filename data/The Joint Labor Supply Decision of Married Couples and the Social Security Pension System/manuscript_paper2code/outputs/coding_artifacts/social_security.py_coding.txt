## social_security.py
import numpy as np
from typing import Union, Dict, Any
from config import Config

class SocialSecurity:
    """
    Implements the Old-Age and Survivors Insurance (OASI) logic for the OLG model.
    
    This class handles the Primary Insurance Amount (PIA) calculation, spousal and 
    survivors benefit rules, and provides marginal benefits for household FOCs.
    It supports both the baseline (current law) and reform (OA only) scenarios.
    """

    def __init__(self, config: Config):
        """
        Initializes the SocialSecurity system with model configuration.

        Args:
            config (Config): Configuration object containing model parameters.
        """
        self.config = config
        # Load parameters from config
        self.mu: float = config.mu
        self.retirement_age: int = config.age_retirement
        self.brackets: np.ndarray = np.array(config.oasi_brackets)
        self.thresholds: np.ndarray = np.array(config.oasi_thresholds)

    def calculate_pia(self, age: int, aime: Union[float, np.ndarray]) -> Union[float, np.ndarray]:
        """
        Calculates the Primary Insurance Amount (PIA) from Average Indexed Monthly Earnings (AIME).
        
        Formula from Section 2.1:
        psi(i, b) = 1{i >= IR} * (1 + mu)**(60 - i) * [0.90*min(b, th1) + 0.32*max(min(b, th2)-th1, 0) + 0.15*max(b-th2, 0)]

        Args:
            age (int): Current age of the individual.
            aime (float or np.ndarray): Average historical earnings (b).
            
        Returns:
            float or np.ndarray: Calculated PIA.
        """
        if age < self.retirement_age:
            return np.zeros_like(aime) if isinstance(aime, np.ndarray) else 0.0

        # Indexing factor for the 60-year-old indexing rule
        indexing_factor = (1.0 + self.mu) ** (60.0 - age)
        
        # Piecewise linear formula
        th1, th2 = self.thresholds
        r1, r2, r3 = self.brackets
        
        # Calculate parts
        part1 = r1 * np.minimum(aime, th1)
        part2 = r2 * np.maximum(np.minimum(aime, th2) - th1, 0.0)
        part3 = r3 * np.maximum(aime - th2, 0.0)
        
        return indexing_factor * (part1 + part2 + part3)

    def calculate_marginal_pia(self, age: int, aime: Union[float, np.ndarray]) -> Union[float, np.ndarray]:
        """
        Calculates the derivative of the PIA with respect to AIME (b).

        Args:
            age (int): Current age of the individual.
            aime (float or np.ndarray): Average historical earnings (b).

        Returns:
            float or np.ndarray: Marginal PIA (psi_b).
        """
        if age < self.retirement_age:
            return np.zeros_like(aime) if isinstance(aime, np.ndarray) else 0.0

        indexing_factor = (1.0 + self.mu) ** (60.0 - age)
        th1, th2 = self.thresholds
        r1, r2, r3 = self.brackets

        # Derivative of piecewise linear function
        if isinstance(aime, np.ndarray):
            m_pia = np.zeros_like(aime)
            m_pia[aime < th1] = r1
            m_pia[(aime >= th1) & (aime < th2)] = r2
            m_pia[aime >= th2] = r3
            return indexing_factor * m_pia
        else:
            if aime < th1:
                return indexing_factor * r1
            elif aime < th2:
                return indexing_factor * r2
            else:
                return indexing_factor * r3

    def get_benefit(self, 
                    m: int, 
                    age: int, 
                    b1: Union[float, np.ndarray], 
                    b2: Union[float, np.ndarray], 
                    psi_t: float = 1.0, 
                    policy_type: str = "baseline") -> Union[float, np.ndarray]:
        """
        Calculates the total household Social Security benefit (tr_SS).
        
        Args:
            m (int): Marital status (0: Married, 1: Single Male, 2: Single Female).
            age (int): Household age.
            b1 (float or np.ndarray): Husband's (or single male's) AIME.
            b2 (float or np.ndarray): Wife's (or single female's) AIME.
            psi_t (float): Benefit adjustment factor (for balancing budget).
            policy_type (str): "baseline" (with spousal/survivors) or "reform" (OA only).
            
        Returns:
            float or np.ndarray: Household OASI benefit.
        """
        pia1 = self.calculate_pia(age, b1)
        pia2 = self.calculate_pia(age, b2)

        if policy_type == "baseline":
            if m == 0:  # Married
                # benefit = psi_t * max(pia1 + pia2, 1.5 * pia1, 1.5 * pia2)
                combined = pia1 + pia2
                spousal1 = 1.5 * pia1
                spousal2 = 1.5 * pia2
                return psi_t * np.maximum(combined, np.maximum(spousal1, spousal2))
            elif m == 1:  # Single Male (or widower)
                # benefit = psi_t * max(pia_self, pia_spouse)
                return psi_t * np.maximum(pia1, pia2)
            elif m == 2:  # Single Female (or widow)
                return psi_t * np.maximum(pia1, pia2)
        
        elif policy_type == "reform":
            if m == 0:  # Married (OA only)
                return psi_t * (pia1 + pia2)
            elif m == 1:  # Single Male
                return psi_t * pia1
            elif m == 2:  # Single Female
                return psi_t * pia2
        
        return np.zeros_like(pia1)

    def get_marginal_benefit(self, 
                             m: int, 
                             age: int, 
                             b1: Union[float, np.ndarray], 
                             b2: Union[float, np.ndarray], 
                             psi_t: float = 1.0, 
                             policy_type: str = "baseline") -> Dict[str, Union[float, np.ndarray]]:
        """
        Calculates the derivatives of the benefit function with respect to b1 and b2.
        Required for Kuhn-Tucker first-order conditions in the household problem.

        Args:
            m (int): Marital status (0, 1, 2).
            age (int): Household age.
            b1 (float or np.ndarray): Husband's AIME.
            b2 (float or np.ndarray): Wife's AIME.
            psi_t (float): Benefit adjustment factor.
            policy_type (str): "baseline" or "reform".

        Returns:
            dict: {"db1": derivative_wrt_b1, "db2": derivative_wrt_b2}
        """
        pia1 = self.calculate_pia(age, b1)
        pia2 = self.calculate_pia(age, b2)
        mpia1 = self.calculate_marginal_pia(age, b1)
        mpia2 = self.calculate_marginal_pia(age, b2)

        res = {"db1": np.zeros_like(mpia1), "db2": np.zeros_like(mpia2)}

        if policy_type == "baseline":
            if m == 0:
                # Logic from Appendix A.1 and Figure 10
                # Marginal wrt b1
                mask_1 = (pia1 >= 0.5 * pia2)
                mask_2 = (pia1 > 2.0 * pia2)
                res["db1"] = psi_t * (mask_1.astype(float) * mpia1 + mask_2.astype(float) * 0.5 * mpia1)
                
                # Marginal wrt b2 (Symmetric)
                mask_3 = (pia2 >= 0.5 * pia1)
                mask_4 = (pia2 > 2.0 * pia1)
                res["db2"] = psi_t * (mask_3.astype(float) * mpia2 + mask_4.astype(float) * 0.5 * mpia2)
                
            elif m == 1: # Single Male (Widower)
                # Max of own or survivors benefit
                mask = (pia1 >= pia2)
                res["db1"] = psi_t * mask.astype(float) * mpia1
                res["db2"] = 0.0 # No longer contributes to wife's historical earnings
                
            elif m == 2: # Single Female (Widow)
                mask = (pia2 >= pia1)
                res["db2"] = psi_t * mask.astype(float) * mpia2
                res["db1"] = 0.0

        elif policy_type == "reform":
            # Simple OA marginals
            if m == 0:
                res["db1"] = psi_t * mpia1
                res["db2"] = psi_t * mpia2
            elif m == 1:
                res["db1"] = psi_t * mpia1
                res["db2"] = 0.0
            elif m == 2:
                res["db1"] = 0.0
                res["db2"] = psi_t * mpia2

        return res

    def get_weighted_benefit_transition(self, 
                                        m: int, 
                                        age: int, 
                                        b1: Union[float, np.ndarray], 
                                        b2: Union[float, np.ndarray], 
                                        psi_t: float, 
                                        weight_baseline: float) -> Union[float, np.ndarray]:
        """
        Calculates a weighted average of baseline and reform benefits for transition cohorts.
        Used for households age 22-60 in year 1 of the policy change.

        Args:
            m (int): Marital status.
            age (int): Household age.
            b1, b2 (float or np.ndarray): AIMEs.
            psi_t (float): Benefit adjustment factor.
            weight_baseline (float): Weight applied to the baseline schedule (omega in Section 4).

        Returns:
            float or np.ndarray: Weighted OASI benefit.
        """
        tr0 = self.get_benefit(m, age, b1, b2, psi_t, policy_type="baseline")
        tr1 = self.get_benefit(m, age, b1, b2, psi_t, policy_type="reform")
        
        return weight_baseline * tr0 + (1.0 - weight_baseline) * tr1
