## utils.py
import numpy as np
from numba import njit
from typing import Tuple

"""
Utility functions for the dynastic Aiyagari-Bewley-Huggett model.
Implements fiscal policy functions, utility calculations, and the intratemporal 
labor supply solver using bisection.
"""

@njit
def tax_function(y: float, lambda_val: float, tau: float) -> float:
    """
    Computes disposable income y^d given gross income y.
    Equation (2): y^d = lambda * y^(1 - tau)
    
    Args:
        y: Gross income (zwh + rk).
        lambda_val: Tax scaling parameter.
        tau: Progressivity parameter.
        
    Returns:
        Disposable income.
    """
    # Use a small epsilon for numerical stability if y is near zero
    y_safe = max(y, 1e-10)
    return lambda_val * (y_safe ** (1.0 - tau))

@njit
def marginal_net_income(y: float, lambda_val: float, tau: float) -> float:
    """
    Computes the derivative of disposable income with respect to gross income.
    dy^d / dy = lambda * (1 - tau) * y^(-tau)
    """
    y_safe = max(y, 1e-10)
    return lambda_val * (1.0 - tau) * (y_safe ** (-tau))

@njit
def marginal_tax_rate(y: float, lambda_val: float, tau: float) -> float:
    """
    Computes the marginal tax rate (MTR).
    MTR = 1 - (dy^d / dy)
    """
    return 1.0 - marginal_net_income(y, lambda_val, tau)

@njit
def utility_c(c: float, sigma: float) -> float:
    """CRRA utility from consumption."""
    if c <= 1e-10:
        return -1e10 # Penalize non-positive consumption
    if sigma == 1.0:
        return np.log(c)
    return (c ** (1.0 - sigma)) / (1.0 - sigma)

@njit
def utility_h(h: float, theta: float, epsilon: float) -> float:
    """Disutility from labor supply."""
    return theta * (h ** (1.0 + epsilon)) / (1.0 + epsilon)

@njit
def marginal_utility_c(c: float, sigma: float) -> float:
    """Derivative of utility with respect to consumption."""
    c_safe = max(c, 1e-10)
    return c_safe ** (-sigma)

@njit
def marginal_disutility_h(h: float, theta: float, epsilon: float) -> float:
    """Derivative of disutility with respect to hours worked."""
    return theta * (h ** epsilon)

@njit
def solve_labor_supply(
    k: float, 
    k_prime: float, 
    z: float, 
    r: float, 
    w: float, 
    lambda_val: float, 
    tau: float,
    sigma: float,
    theta: float,
    epsilon: float,
    tol: float = 1e-8,
    max_iter: int = 100
) -> Tuple[float, float]:
    """
    Solves for optimal labor supply h using the bisection method on the FOC:
    f(h) = theta * h^epsilon - c^(-sigma) * [dy^d/dy * dy/dh] = 0
    
    where:
    y = zwh + rk
    c = y^d(y) + k - k_prime
    dy/dh = zw
    
    Args:
        k: Current period capital.
        k_prime: Next period capital (savings).
        z: Labor efficiency.
        r: Interest rate.
        w: Wage rate.
        lambda_val: Tax scaling factor.
        tau: Tax progressivity.
        sigma, theta, epsilon: Preference parameters.
        
    Returns:
        (h_opt, c_opt): Optimal hours worked and corresponding consumption.
    """
    # Search range for h in [0, 1]
    h_low = 1e-7
    h_high = 1.0 - 1e-7
    
    def get_foc_val(h_val: float) -> float:
        y = z * w * h_val + r * k
        y_disp = tax_function(y, lambda_val, tau)
        c = y_disp + k - k_prime
        
        if c <= 0:
            # Consumption is negative; labor must increase or k_prime must decrease.
            # Returning a large negative value to force h higher in bisection.
            return -1e10
        
        m_net = marginal_net_income(y, lambda_val, tau)
        return marginal_disutility_h(h_val, theta, epsilon) - \
               marginal_utility_c(c, sigma) * m_net * z * w

    # Check boundaries for bisection
    f_low = get_foc_val(h_low)
    f_high = get_foc_val(h_high)
    
    # If the interval doesn't bracket zero, return corner solutions
    if f_low >= 0:
        h_opt = h_low
    elif f_high <= 0:
        h_opt = h_high
    else:
        # Bisection loop
        for _ in range(max_iter):
            h_mid = 0.5 * (h_low + h_high)
            f_mid = get_foc_val(h_mid)
            
            if abs(f_mid) < tol or (h_high - h_low) < tol:
                h_opt = h_mid
                break
            
            if f_mid > 0:
                h_high = h_mid
            else:
                h_low = h_mid
        else:
            h_opt = 0.5 * (h_low + h_high)

    # Compute final consumption for the chosen h_opt
    y_final = z * w * h_opt + r * k
    c_final = tax_function(y_final, lambda_val, tau) + k - k_prime
    
    return h_opt, max(c_final, 1e-10)

@njit
def calculate_certainty_equivalent_c(v_c: float, sigma: float, beta: float) -> float:
    """
    Solves for certainty equivalent consumption (c_tilde) from Equation (7).
    v_c = (1 / (1 - beta)) * (c_tilde^(1 - sigma) / (1 - sigma))
    
    Args:
        v_c: Lifetime value derived from consumption only.
        sigma: CRRA coefficient.
        beta: Discount factor.
    """
    val_norm = v_c * (1.0 - beta) * (1.0 - sigma)
    # Ensure argument for power is positive
    if val_norm <= 0 and sigma < 1:
         return 0.0
    return val_norm ** (1.0 / (1.0 - sigma))

@njit
def calculate_certainty_equivalent_h(v_l: float, theta: float, epsilon: float, beta: float) -> float:
    """
    Solves for certainty equivalent labor (h_tilde) from Equation (7).
    v_l = (1 / (1 - beta)) * theta * (h_tilde^(1 + epsilon) / (1 + epsilon))
    
    Args:
        v_l: Lifetime value (disutility) derived from labor only.
        theta: Disutility weight.
        epsilon: Labor curvature.
        beta: Discount factor.
    """
    # h_tilde = [ (v_l * (1 - beta) * (1 + epsilon)) / theta ] ^ (1 / (1 + epsilon))
    val_norm = (v_l * (1.0 - beta) * (1.0 + epsilon)) / theta
    return max(val_norm, 0.0) ** (1.0 / (1.0 + epsilon))

@njit
def get_gini(x: np.ndarray, weights: np.ndarray) -> float:
    """
    Calculates the Gini coefficient for a weighted distribution.
    
    Args:
        x: Values (e.g., income, wealth).
        weights: Corresponding mass/density for each value.
        
    Returns:
        Gini coefficient.
    """
    # Sort by values
    idx = np.argsort(x)
    sorted_x = x[idx]
    sorted_w = weights[idx]
    
    # Cumulative weight and cumulative value
    sum_w = np.sum(sorted_w)
    cum_w = np.cumsum(sorted_w) / sum_w
    cum_xw = np.cumsum(sorted_x * sorted_w)
    cum_xw /= cum_xw[-1]
    
    # Area under Lorenz curve (trapezoidal rule)
    # Gini = 1 - 2 * sum( (cum_w_i - cum_w_{i-1}) * (cum_xw_i + cum_xw_{i-1})/2 )
    # This formula approximates the Gini coefficient for discrete distributions
    area = 0.0
    for i in range(1, len(x)):
        area += (cum_w[i] - cum_w[i-1]) * (cum_xw[i] + cum_xw[i-1])
        
    return 1.0 - area
